cmake_minimum_required(VERSION 3.20)
project(CompositorQtProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Widgets)
add_subdirectory(../native native)
add_executable(qt_probe probe.cpp scene.cpp composite.cpp brush.cpp brush_probe.cpp brush_performance.cpp text_probe.cpp window.cpp)
target_link_libraries(qt_probe PRIVATE Qt6::Widgets compositor_native)
if(WIN32)
    target_link_libraries(qt_probe PRIVATE psapi)
endif()
if(MSVC)
    target_compile_options(qt_probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(qt_probe PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_executable(qt_s02_checks brush_performance_checks.cpp brush_performance.cpp brush.cpp scene.cpp composite.cpp)
target_link_libraries(qt_s02_checks PRIVATE Qt6::Widgets compositor_native)
if(WIN32)
    target_link_libraries(qt_s02_checks PRIVATE psapi)
endif()
if(MSVC)
    target_compile_options(qt_s02_checks PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(qt_s02_checks PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME qt_s02_lifecycle COMMAND qt_s02_checks "${CMAKE_CURRENT_SOURCE_DIR}/../../../docs/windows/fixtures/brush")
set_tests_properties(qt_s02_lifecycle PROPERTIES TIMEOUT 180)

qt_add_resources(qt_probe "text_fonts" PREFIX "/fonts"
    BASE "${CMAKE_CURRENT_SOURCE_DIR}/../../../Compositor/Resources/Fonts"
    FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../../Compositor/Resources/Fonts/SourceHanSansSC-Regular.otf")
configure_file("../../../Compositor/Resources/Licenses/SourceHanSans-OFL.txt"
    "Licenses/SourceHanSans-OFL.txt" COPYONLY)
