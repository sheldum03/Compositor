cmake_minimum_required(VERSION 3.20)
project(CompositorHeicProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(libheif 1.23.4 EXACT CONFIG REQUIRED)
add_executable(heic_probe probe.cpp)
target_link_libraries(heic_probe PRIVATE heif)
if(MSVC)
    target_compile_options(heic_probe PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(heic_probe PRIVATE -Wall -Wextra -Werror)
endif()
