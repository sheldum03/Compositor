cmake_minimum_required(VERSION 3.20)
project(CompositorAiProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ORT_ROOT "" CACHE PATH "Extracted official ONNX Runtime 1.30.0 CPU SDK")
if(NOT EXISTS "${ORT_ROOT}/include/onnxruntime_cxx_api.h")
    message(FATAL_ERROR "Set ORT_ROOT to the verified ONNX Runtime SDK")
endif()
find_library(ORT_LIBRARY NAMES onnxruntime PATHS "${ORT_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_executable(ai_probe probe.cpp)
add_executable(birefnet_probe probe.cpp)
target_compile_definitions(birefnet_probe PRIVATE COMPOSITOR_BIREFNET)
foreach(probe IN ITEMS ai_probe birefnet_probe)
    target_include_directories(${probe} SYSTEM PRIVATE "${ORT_ROOT}/include")
    target_link_libraries(${probe} PRIVATE "${ORT_LIBRARY}" Threads::Threads)
    if(MSVC)
        target_compile_options(${probe} PRIVATE /W4 /WX /utf-8)
    else()
        target_compile_options(${probe} PRIVATE -Wall -Wextra -Werror)
    endif()

    if(MINGW)
        target_link_options(${probe} PRIVATE -municode)
    endif()
endforeach()
